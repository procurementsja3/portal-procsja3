Google Apps Script - Data Ceklist Warehouse

Skrip sinkronisasi 16 Kolom Spreadsheet (Timestamp s/d Email Address)

Langkah Pasang di Google Spreadsheet:
  1. Buka Spreadsheet target Google Sheet di browser.
  2. Klik menu atas: Ekstensi (Extensions) → Apps Script.
  3. Hapus kode bawaan di file Code.gs, lalu Paste (Tempel) kode di bawah ini.
  4. Klik tombol Deploy (Terapkan) di pojok kanan atas → pilih New deployment (Penerapan baru).
  5. Pilih jenis Web app (Aplikasi web):
    • Execute as: Me (Saya)
    • Who has access: Anyone (Siapa saja)
  6. Klik Deploy, izinkan akses (Authorize), lalu salin Web App URL yang dihasilkan.
  7. Paste URL tersebut di menu "Link Sheet" pada SIBIKOM untuk sinkronisasi otomatis!
Urutan Kolom Terintegrasi (A - P):

A: Timestamp | B: Nama Warehouse Inspektor | C: Tanggal Inspeksi | D: Bagian | E: Area | F: Nomor gudang | G: Kategori Audit | H: Tingkat Keparahan | I: Dokumentasi foto 1 | J: Dokumentasi foto 2 | K: Diskripsi Temuan | L: Tindakan Perbaikan | M: PIC | N: Target Selesai | O: Status | P: Email Address

File: Code.gs 
/**
 * =========================================================================
 * SIBIKOM - Google Apps Script untuk "Data Ceklist Warehouse"
 * PT Santos Jaya Abadi 3 (SJA3) - Gudang & Supply Chain
 * =========================================================================
 */

var TARGET_SHEET_NAME = "DATA_CEKLIST_WAREHOUSE";

/**
 * 1. doGet: Mengirimkan data inspeksi dalam format JSON ke aplikasi SIBIKOM
 */
function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = getCeklistSheet(ss, e);

    if (!sheet) {
      return responseJSON({
        status: "error",
        message: "Sheet tidak ditemukan. Silakan jalankan setupCeklistSheet() terlebih dahulu."
      });
    }

    var data = readSheetData(sheet);

    // Support callback JSONP jika dipanggil dengan ?callback=
    var callback = e && e.parameter && e.parameter.callback;
    if (callback) {
      return ContentService.createTextOutput(callback + "(" + JSON.stringify({
        status: "success",
        total: data.length,
        timestamp: new Date().toISOString(),
        data: data
      }) + ")").setMimeType(ContentService.MimeType.JAVASCRIPT);
    }

    return responseJSON({
      status: "success",
      total: data.length,
      timestamp: new Date().toISOString(),
      data: data
    });

  } catch (err) {
    return responseJSON({ status: "error", message: err.toString() });
  }
}

/**
 * 2. doPost: Menerima input / penambahan data inspeksi baru dari aplikasi SIBIKOM
 */
function doPost(e) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000);

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = getCeklistSheet(ss, e);
    if (!sheet) {
      sheet = setupCeklistSheet();
    }

    var p = e.parameter || {};
    if (e.postData && e.postData.contents) {
      try {
        var jsonBody = JSON.parse(e.postData.contents);
        for (var key in jsonBody) { p[key] = jsonBody[key]; }
      } catch (ex) {}
    }

    var timestamp = p.timestamp || p.waktu || Utilities.formatDate(new Date(), "Asia/Jakarta", "M/d/yyyy HH:mm:ss");
    var inspector = p.petugas || p.namaWarehouseInspektor || p.nama || "";
    var tglInspeksi = p.tanggal || p.tanggalInspeksi || Utilities.formatDate(new Date(), "Asia/Jakarta", "M/d/yyyy");
    var bagian = p.bagian || "GBB";
    var area = p.area || "";
    var nomorGudang = p.nomorGudang || p.nomor_gudang || "";
    var kategoriAudit = p.kategoriAudit || p.kategori || "5 R";
    var tingkatKeparahan = p.tingkatKeparahan || p.keparahan || "Low";
    var foto1 = normalizeDriveUrl(p.foto1 || p.dokumentasiFoto1 || "");
    var foto2 = normalizeDriveUrl(p.foto2 || p.dokumentasiFoto2 || "");
    var deskripsi = p.deskripsi || p.diskripsiTemuan || "";
    var tindakan = p.tindakan || p.tindakanPerbaikan || "";
    var pic = p.pic || "";
    var targetSelesai = p.targetSelesai || "";
    var status = p.status || "Open";
    var email = p.email || p.emailAddress || "";

    var newRow = [
      timestamp,        // A: Timestamp
      inspector,        // B: Nama Warehouse Inspektor
      tglInspeksi,      // C: Tanggal Inspeksi
      bagian,           // D: Bagian 
      area,             // E: Area
      nomorGudang,      // F: Nomor gudang
      kategoriAudit,    // G: Kategori Audit 
      tingkatKeparahan, // H: Tingkat Keparahan
      foto1,            // I: Dokumentasi foto 1
      foto2,            // J: Dokumentasi foto 2
      deskripsi,        // K: Diskripsi Temuan
      tindakan,         // L: Tindakan Perbaikan
      pic,              // M: PIC
      targetSelesai,    // N: Target Selesai 
      status,           // O: Status
      email             // P: Email Address
    ];

    sheet.appendRow(newRow);

    return responseJSON({
      status: "success",
      message: "Data Ceklist Warehouse berhasil dicatat ke Google Sheet.",
      row: newRow
    });

  } catch (err) {
    return responseJSON({ status: "error", message: err.toString() });
  } finally {
    lock.releaseLock();
  }
}

/**
 * 3. Membaca data baris dan memetakan menjadi JSON
 */
function readSheetData(sheet) {
  var lastRow = sheet.getLastRow();
  var lastCol = Math.max(sheet.getLastColumn(), 16);
  if (lastRow < 2) return [];

  var values = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();
  var result = [];

  for (var i = 0; i < values.length; i++) {
    var r = values[i];
    var timestampStr = formatCellValue(r[0]);
    var inspector = formatCellValue(r[1]);
    var tglStr = formatCellValue(r[2]);

    if (!inspector && !timestampStr && !tglStr) continue;

    result.push({
      id: "CW-" + (1001 + i),
      waktu: timestampStr,
      petugas: inspector,
      namaWarehouseInspektor: inspector,
      tanggal: tglStr,
      tanggalInspeksi: tglStr,
      bagian: formatCellValue(r[3]),
      area: formatCellValue(r[4]),
      nomorGudang: formatCellValue(r[5]),
      kategoriAudit: formatCellValue(r[6]),
      tingkatKeparahan: formatCellValue(r[7]),
      foto1: normalizeDriveUrl(formatCellValue(r[8])),
      foto2: normalizeDriveUrl(formatCellValue(r[9])),
      deskripsi: formatCellValue(r[10]),
      diskripsiTemuan: formatCellValue(r[10]),
      tindakan: formatCellValue(r[11]),
      tindakanPerbaikan: formatCellValue(r[11]),
      pic: formatCellValue(r[12]),
      targetSelesai: formatCellValue(r[13]),
      status: formatCellValue(r[14]) || "Open",
      email: formatCellValue(r[15])
    });
  }

  return result;
}

/**
 * 4. Mendapatkan Tab Sheet Ceklist yang Tepat
 */
function getCeklistSheet(ss, e) {
  var requestedName = (e && e.parameter && e.parameter.sheet) ? e.parameter.sheet : TARGET_SHEET_NAME;
  var sheet = ss.getSheetByName(requestedName);
  if (sheet) return sheet;

  // Deteksi otomatis jika sheet bernama lain (misal Form Responses 1)
  var sheets = ss.getSheets();
  for (var i = 0; i < sheets.length; i++) {
    var s = sheets[i];
    if (s.getLastRow() >= 1) {
      var headerRow = s.getRange(1, 1, 1, Math.min(s.getLastColumn(), 16)).getValues()[0];
      var headerStr = headerRow.join(" ").toLowerCase();
      if (headerStr.indexOf("inspektor") !== -1 || headerStr.indexOf("temuan") !== -1 || headerStr.indexOf("kategori audit") !== -1) {
        return s;
      }
    }
  }
  return sheets[0];
}

/**
 * 5. Inisialisasi Header Standar jika Sheet Kosong
 */
function setupCeklistSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(TARGET_SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(TARGET_SHEET_NAME);
  }

  var headers = [
    "Timestamp",
    "Nama Warehouse Inspektor",
    "Tanggal Inspeksi",
    "Bagian ",
    "Area",
    "Nomor gudang",
    "Kategori Audit ",
    "Tingkat Keparahan",
    "Dokumentasi foto 1",
    "Dokumentasi foto 2",
    "Diskripsi Temuan",
    "Tindakan Perbaikan",
    "PIC",
    "Target Selesai ",
    "Status",
    "Email Address"
  ];

  var headerRange = sheet.getRange(1, 1, 1, headers.length);
  headerRange.setValues([headers]);
  headerRange.setFontWeight("bold");
  headerRange.setBackground("#1e293b");
  headerRange.setFontColor("#ffffff");
  sheet.setFrozenRows(1);
  return sheet;
}

/**
 * 6. Format URL Google Drive
 */
function normalizeDriveUrl(url) {
  if (!url || typeof url !== "string") return "";
  url = url.trim();
  var match = url.match(/[?&]id=([a-zA-Z0-9_-]+)/) || url.match(/\/d\/([a-zA-Z0-9_-]+)/);
  if (match && match[1]) {
    return "https://drive.google.com/open?id=" + match[1];
  }
  return url;
}

function formatCellValue(val) {
  if (val === null || val === undefined) return "";
  if (val instanceof Date) {
    return Utilities.formatDate(val, "Asia/Jakarta", "M/d/yyyy HH:mm:ss");
  }
  return String(val).trim();
}

function responseJSON(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

/**
 * 7. Menu Cepat di Spreadsheet
 */
function onOpen() {
  var ui = SpreadsheetApp.getUi();
  ui.createMenu("⚡ SIBIKOM Warehouse")
    .addItem("🔍 Inisialisasi Header Standar", "setupCeklistSheet")
    .addItem("🧪 Tes Data Ceklist", "testReadData")
    .addToUi();
}

function testReadData() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = getCeklistSheet(ss, null);
  var data = readSheetData(sheet);
  SpreadsheetApp.getUi().alert("Berhasil membaca " + data.length + " baris data inspeksi Ceklist Warehouse.");
}
